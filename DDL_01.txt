-- ============================================================================
-- MODELLO FISICO / DDL POSTGRESQL (ER 1.5)
-- ============================================================================

-- ============================================================================
-- 1. AREA TOPOGRAFICA
-- ============================================================================

CREATE TABLE sito (
    id SERIAL PRIMARY KEY,
    localita VARCHAR(150) NOT NULL,
    stato VARCHAR(50) NOT NULL,
    data_inizio DATE NOT NULL,
    data_fine DATE,
    note TEXT,
    -- GIS: In futuro abilitare PostGIS e aggiungere
    -- coordinate_geografiche
    CONSTRAINT chk_sito_date CHECK (data_fine IS NULL OR data_fine >= data_inizio)
);

CREATE TABLE cantiere (
    id SERIAL PRIMARY KEY,
    codice VARCHAR(50) NOT NULL UNIQUE,
    descrizione TEXT,
    data_inizio DATE NOT NULL,
    data_fine DATE,
    note TEXT,
    sito_id INT NOT NULL,
    -- GIS: In futuro abilitare PostGIS e aggiungere:
    -- poligono
    -- fotopiano
    CONSTRAINT fk_cantiere_sito FOREIGN KEY (sito_id) REFERENCES sito(id) ON DELETE CASCADE,
    CONSTRAINT chk_cantiere_date CHECK (data_fine IS NULL OR data_fine >= data_inizio)
);

CREATE TABLE quadrante (
    cantiere_id INT NOT NULL,
    x INT NOT NULL,
    y INT NOT NULL,
    -- GIS: In futuro abilitare PostGIS e aggiungere:
    -- poligono
    PRIMARY KEY (cantiere_id, x, y),
    CONSTRAINT fk_quadrante_cantiere FOREIGN KEY (cantiere_id) REFERENCES cantiere(id) ON DELETE CASCADE
);

-- ============================================================================
-- 2. AREA ORGANIZZATIVA
-- ============================================================================

CREATE TABLE archeologo (
    id SERIAL PRIMARY KEY,
    email VARCHAR(150) NOT NULL UNIQUE,
    nome VARCHAR(80) NOT NULL,
    cognome VARCHAR(80) NOT NULL,
    qualifica VARCHAR(100),
    ente_affiliazione VARCHAR(150),
    note TEXT
);

CREATE TABLE missione (
    id SERIAL PRIMARY KEY,
    codice VARCHAR(50) NOT NULL UNIQUE,
    descrizione TEXT,
    data_inizio DATE NOT NULL,
    data_fine DATE,
    cantiere_id INT NOT NULL,
    apre BOOLEAN DEFAULT FALSE,
    chiude BOOLEAN DEFAULT FALSE,
    CONSTRAINT fk_missione_cantiere FOREIGN KEY (cantiere_id) REFERENCES cantiere(id) ON DELETE RESTRICT,
    CONSTRAINT chk_missione_date CHECK (data_fine IS NULL OR data_fine >= data_inizio)
);

-- Tabella di collegamento molti-a-molti: Missione - Archeologo (ha partecipato)
CREATE TABLE missione_archeologo (
    missione_id INT NOT NULL,
    archeologo_id INT NOT NULL,
    PRIMARY KEY (missione_id, archeologo_id),
    CONSTRAINT fk_ma_missione FOREIGN KEY (missione_id) REFERENCES missione(id) ON DELETE CASCADE,
    CONSTRAINT fk_ma_archeologo FOREIGN KEY (archeologo_id) REFERENCES archeologo(id) ON DELETE CASCADE
);

-- ============================================================================
-- 3. AREA STRATIGRAFICA (HARRIS)
-- ============================================================================

CREATE TABLE unita_stratigrafica (
    id SERIAL PRIMARY KEY,
    cantiere_id INT NOT NULL,
    numero INT NOT NULL,
    data DATE NOT NULL,
    tipo VARCHAR(50) NOT NULL,
    stato_conservazione VARCHAR(50),
    quota_superiore NUMERIC(7, 2),
    quota_inferiore NUMERIC(7, 2),
    edificio VARCHAR(100),
    stanza VARCHAR(100),
    fase VARCHAR(50),
    descrizione TEXT,
    interpretazione TEXT,
    archeologo_compilatore_id INT NOT NULL,
    -- GIS: In futuro abilitare PostGIS e aggiungere
    -- coordinate_geografiche
    -- poligono
    CONSTRAINT uq_cantiere_numero_us UNIQUE (cantiere_id, numero),
    CONSTRAINT fk_us_cantiere FOREIGN KEY (cantiere_id) REFERENCES cantiere(id) ON DELETE RESTRICT,
    CONSTRAINT fk_us_archeologo FOREIGN KEY (archeologo_compilatore_id) REFERENCES archeologo(id) ON DELETE RESTRICT
);

-- Tabella per la relazione esplicita 'localizza' (Quadrante - Unità Stratigrafica)
CREATE TABLE unita_stratigrafica_quadrante (
    unita_stratigrafica_id INT NOT NULL,
    cantiere_id INT NOT NULL,
    quadrante_x INT NOT NULL,
    quadrante_y INT NOT NULL,
    PRIMARY KEY (unita_stratigrafica_id, cantiere_id, quadrante_x, quadrante_y),
    CONSTRAINT fk_usq_us FOREIGN KEY (unita_stratigrafica_id) 
        REFERENCES unita_stratigrafica(id) ON DELETE CASCADE,
    CONSTRAINT fk_usq_quadrante FOREIGN KEY (cantiere_id, quadrante_x, quadrante_y) 
        REFERENCES quadrante(cantiere_id, x, y) ON DELETE CASCADE
);

-- Relazioni stratigrafiche (Harris Matrix)
CREATE TABLE relazione_stratigrafica (
    id SERIAL PRIMARY KEY,
    us_origine_id INT NOT NULL,
    us_destinazione_id INT NOT NULL,
    tipo VARCHAR(50) NOT NULL, -- Copre/Coperto da, Riempie/Riempito da, Taglia/Tagliato da, Si appoggia a/Gli si appoggia, ecc.
    CONSTRAINT fk_rel_us_origine FOREIGN KEY (us_origine_id) REFERENCES unita_stratigrafica(id) ON DELETE CASCADE,
    CONSTRAINT fk_rel_us_destinazione FOREIGN KEY (us_destinazione_id) REFERENCES unita_stratigrafica(id) ON DELETE CASCADE,
    CONSTRAINT chk_evita_autoriferimento CHECK (us_origine_id <> us_destinazione_id),
    CONSTRAINT uq_relazione_us UNIQUE (us_origine_id, us_destinazione_id, tipo)
);

-- ============================================================================
-- 4. AREA DOCUMENTALE
-- ============================================================================

CREATE TABLE media (
    id SERIAL PRIMARY KEY,
    unita_stratigrafica_id INT NOT NULL,
    uri VARCHAR(500) NOT NULL,
    data_caricamento TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    tipo_media VARCHAR(20) NOT NULL, -- 'FOTO', 'VIDEO', 'DISEGNO'
    tipo_disegno VARCHAR(50),        -- 'Pianta', 'Sezione' (valorizzato solo se tipo_media = 'DISEGNO')
    CONSTRAINT fk_media_us FOREIGN KEY (unita_stratigrafica_id) REFERENCES unita_stratigrafica(id) ON DELETE CASCADE,
    CONSTRAINT chk_tipo_media CHECK (tipo_media IN ('FOTO', 'VIDEO', 'DISEGNO')),
    CONSTRAINT chk_tipo_disegno CHECK (
        (tipo_media = 'DISEGNO' AND tipo_disegno IN ('Pianta', 'Sezione'))
        OR
        (tipo_media <> 'DISEGNO' AND tipo_disegno IS NULL)
    )
);
